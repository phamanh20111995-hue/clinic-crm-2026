MANAGEMENT_ROLES = frozenset({'QUAN_LY', 'CHU_DN'})
PAYMENT_ROLES = {'KE_TOAN'} | MANAGEMENT_ROLES


def _num(v):
    try:
        return int(round(float(v or 0)))
    except (TypeError, ValueError):
        return 0


def _next_disbursement_date(start):
    """T+1, bo qua Thu 6 (4) / Thu 7 (5) / CN (6)."""
    d = start + timedelta(days=1)
    while d.weekday() in (4, 5, 6):
        d += timedelta(days=1)
    return d

def _recompute_payment_status(contract):
    paid = sum(p.total for p in contract.payments.all())
    if paid <= 0:
        contract.payment_status = 'pending'
    elif paid >= contract.final_amount:
        contract.payment_status = 'received'
    else:
        contract.payment_status = 'partial'
    contract.save(update_fields=['payment_status'])


@api_view(['GET', 'POST'])
@permission_classes([IsAuthenticated])
def contract_payments(request, pk):
    """GET: lich su thu cua HD. POST: thu tien (TM + CK + tra gop)."""
    contract = Contract.objects.filter(pk=pk, is_deleted=False).first()
    if not contract:
        return Response({'detail': 'Khong tim thay hop dong.'}, status=404)

    if request.method == 'GET':
        qs = contract.payments.all().prefetch_related('installments')
        return Response(PaymentOutSerializer(qs, many=True).data)

    # POST -> thu tien (chi KE_TOAN / QUAN_LY / CHU_DN)
    if getattr(request.user, 'role', None) not in PAYMENT_ROLES:
        return Response({'detail': 'Ban khong co quyen thu tien.'}, status=403)

    data = request.data or {}
    cash = _num(data.get('cash_amount'))
    transfer = _num(data.get('transfer_amount'))
    inst_amount = _num(data.get('installment_amount'))
    if cash < 0 or transfer < 0 or inst_amount < 0:
        return Response({'detail': 'So tien khong hop le.'}, status=400)
    total = cash + transfer + inst_amount
    if total <= 0:
        return Response({'detail': 'Tong tien thu phai lon hon 0.'}, status=400)

    received_at = data.get('received_at') or timezone.localdate().isoformat()
    try:
        r_date = datetime.strptime(received_at, '%Y-%m-%d').date()
    except (ValueError, TypeError):
        r_date = timezone.localdate()

    inst_payload = data.get('installment') or None
    if inst_amount > 0 and not inst_payload:
        return Response({'detail': 'Thieu thong tin tra gop.'}, status=400)

    with transaction.atomic():
        payment = Payment.objects.create(
            contract=contract,
            total=total,
            cash_amount=cash,
            transfer_amount=transfer,
            installment_amount=inst_amount,
            received_at=r_date,
            received_by=request.user,
            note=data.get('note', ''),
        )

        if inst_amount > 0:
            provider_type = inst_payload.get('provider_type', 'vay_tieu_dung')
            principal = _num(inst_payload.get('principal')) or inst_amount
            conversion_fee = _num(inst_payload.get('conversion_fee'))
            interest = _num(inst_payload.get('interest'))
            fee_bearer = inst_payload.get('fee_bearer', 'khach')
            interest_bearer = inst_payload.get('interest_bearer', 'khach')

            # Tach phi chuyen doi theo ben chiu
            if fee_bearer == 'pk':
                fee_pk, fee_khach = conversion_fee, 0
            elif fee_bearer == 'khach':
                fee_pk, fee_khach = 0, conversion_fee
            else:  # chia
                fee_pk = _num(inst_payload.get('fee_pk'))
                fee_khach = _num(inst_payload.get('fee_khach'))

            # Tach lai theo ben chiu (PK chiu lai = hoan lai cho khach)
            if interest_bearer == 'pk':
                interest_pk, interest_khach = interest, 0
            elif interest_bearer == 'khach':
                interest_pk, interest_khach = 0, interest
            else:  # chia
                interest_pk = _num(inst_payload.get('interest_pk'))
                interest_khach = _num(inst_payload.get('interest_khach'))

            pk_cost = fee_pk + interest_pk
            pk_net = principal - pk_cost

            if provider_type == 'the_tin_dung':
                inst_status = 'da_ve'
                expected_date = r_date
                actual_date = r_date
            else:  # vay_tieu_dung -> cho giai ngan
                inst_status = 'cho_giai_ngan'
                expected_date = _next_disbursement_date(r_date)
                actual_date = None

            Installment.objects.create(
                payment=payment,
                provider=inst_payload.get('provider', ''),
                provider_type=provider_type,
                principal=principal,
                conversion_fee=conversion_fee,
                fee_bearer=fee_bearer,
                fee_pk=fee_pk,
                fee_khach=fee_khach,
                interest=interest,
                interest_bearer=interest_bearer,
                interest_pk=interest_pk,
                interest_khach=interest_khach,
                pk_net=pk_net,
                pk_cost=pk_cost,
                status=inst_status,
                expected_date=expected_date,
                actual_date=actual_date,
            )

        _recompute_payment_status(contract)

    return Response(PaymentOutSerializer(payment).data, status=201)


@api_view(['GET'])
@permission_classes([IsAuthenticated])
def cong_no_list(request):
    """Danh sach HD da duyet con du no (final_amount - tong da thu > 0)."""
    qs = (Contract.objects
          .filter(is_deleted=False, approval_status='approved')
          .select_related('customer')
          .prefetch_related('payments'))
    data = []
    for c in qs:
        paid = sum(p.total for p in c.payments.all())
        remaining = int(c.final_amount) - int(paid)
        if remaining <= 0:
            continue
        data.append({
            'contract_id': c.id,
            'contract_no': c.contract_no,
            'customer_name': c.customer.full_name,
            'customer_phone': c.customer.phone,
            'final_amount': int(c.final_amount),
            'paid': int(paid),
            'remaining': remaining,
            'created_at': c.created_at,
        })
    return Response(data)


@api_view(['GET'])
@permission_classes([IsAuthenticated])
def installment_list(request):
    """Theo doi khoan tra gop. Loc ?status= & ?provider_type="""
    qs = (Installment.objects
          .select_related('payment', 'payment__contract', 'payment__contract__customer')
          .all())
    status_f = request.query_params.get('status')
    if status_f:
        qs = qs.filter(status=status_f)
    provider_type = request.query_params.get('provider_type')
    if provider_type:
        qs = qs.filter(provider_type=provider_type)
    data = []
    for inst in qs:
        row = InstallmentOutSerializer(inst).data
        c = inst.payment.contract
        row['contract_id'] = c.id
        row['contract_no'] = c.contract_no
        row['customer_name'] = c.customer.full_name
        data.append(row)
    return Response(data)


@api_view(['POST'])
@permission_classes([IsAuthenticated])
def confirm_installment(request, pk):
    """Xac nhan 1 khoan tra gop da ve ket."""
    if getattr(request.user, 'role', None) not in PAYMENT_ROLES:
        return Response({'detail': 'Ban khong co quyen xac nhan.'}, status=403)
    inst = Installment.objects.filter(pk=pk).first()
    if not inst:
        return Response({'detail': 'Khong tim thay khoan tra gop.'}, status=404)
    actual = request.data.get('actual_date') or timezone.localdate().isoformat()
    inst.status = 'da_ve'
    inst.actual_date = actual
    inst.save(update_fields=['status', 'actual_date'])
    return Response(InstallmentOutSerializer(inst).data)

@permission_classes([IsAuthenticated])
def confirm_installments_batch(request):
    """Xac nhan ve ket theo lo. Body: {"ids": [...], "actual_date": "YYYY-MM-DD"}"""
    if getattr(request.user, 'role', None) not in PAYMENT_ROLES:
        return Response({'detail': 'Ban khong co quyen xac nhan.'}, status=403)
    ids = request.data.get('ids') or []
    if not isinstance(ids, list) or not ids:
        return Response({'detail': 'Thieu danh sach id.'}, status=400)
    actual = request.data.get('actual_date') or timezone.localdate().isoformat()
    updated = Installment.objects.filter(pk__in=ids).update(status='da_ve', actual_date=actual)
    return Response({'updated': updated})

def _contract_queryset(user):
    """
    RBAC:
    - QUAN_LY / CHU_DN      → tất cả
    - KE_TOAN               → tất cả (cần duyệt)
    - LEAD_SALE             → tất cả của team Sale
    - SALE                  → HĐ do mình tạo
    - CSKH / LEAD_CSKH      → HĐ KH của mình
    """
    qs = Contract.objects.filter(is_deleted=False).select_related(
        'customer', 'created_by', 'approved_by', 'appointment'
    )
    if user.role in MANAGEMENT_ROLES or user.role in ('KE_TOAN', 'LEAD_SALE'):
        return qs
    if user.role == 'SALE':
        return qs.filter(created_by=user)
    if user.role in ('CSKH', 'LEAD_CSKH'):
        return qs.filter(customer__sale__isnull=False) | qs.filter(customer__tele=user)
    return qs.none()


class ContractListCreateView(generics.ListCreateAPIView):
    """
    GET  /api/contracts/      — Danh sách HĐ (filter theo role)
    POST /api/contracts/      — Tạo HĐ nháp (SALE / CSKH / Management)
    Query: approval_status, customer_id, payment_status
    """
    permission_classes = [IsAuthenticated]

    def get_serializer_class(self):
        return ContractCreateSerializer if self.request.method == 'POST' else ContractListSerializer

    def get_queryset(self):
        qs = _contract_queryset(self.request.user)
        p = self.request.query_params
        if p.get('approval_status'): qs = qs.filter(approval_status=p['approval_status'])
        if p.get('payment_status'):  qs = qs.filter(payment_status=p['payment_status'])
        if p.get('customer_id'):     qs = qs.filter(customer_id=p['customer_id'])
        return qs.order_by('-created_at')

    def create(self, request, *args, **kwargs):
        allowed = {'SALE', 'CSKH', 'LEAD_SALE', 'LEAD_CSKH'} | MANAGEMENT_ROLES
        if request.user.role not in allowed:
            return Response({'detail': 'Không có quyền tạo hợp đồng.'}, status=403)
        return super().create(request, *args, **kwargs)

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def submit_contract(request, pk):
    """
    POST /api/contracts/{id}/submit/
    Sale/CSKH submit HĐ lên KT duyệt.
    draft → pending_kt
    """
    allowed = {'SALE', 'CSKH', 'LEAD_SALE', 'LEAD_CSKH'} | MANAGEMENT_ROLES
    if request.user.role not in allowed:
        return Response({'detail': 'Không có quyền submit.'}, status=403)

    contract = _contract_queryset(request.user).filter(pk=pk).first()
    if not contract:
        return Response({'detail': 'Không tìm thấy HĐ.'}, status=404)
    if contract.approval_status != 'draft':
        return Response({'detail': f'HĐ đang ở trạng thái {contract.get_approval_status_display()}, không thể submit.'}, status=400)

    contract.approval_status = 'pending_kt'
    contract.save(update_fields=['approval_status'])

    # Cập nhật customer status
    contract.customer.status = 'dat_lich'
    contract.customer.save(update_fields=['status', 'updated_at'])

    send_notification_to_roles(
        ['KE_TOAN'],
        'hd_pending_kt',
        f'HĐ {contract.contract_no} cần duyệt',
        body=f'KH: {contract.customer.full_name} — {contract.final_amount:,.0f} VNĐ',
        data={'contract_id': contract.id},
    )
    return Response(ContractDetailSerializer(contract).data)


@api_view(['POST'])
@permission_classes([IsAuthenticated])
def approve_contract(request, pk):
    """
    POST /api/contracts/{id}/approve/
    KT duyệt HĐ → pending_kt → approved.
    Ghi nhận approved_by, approved_at.
    """
    if request.user.role not in {'KE_TOAN'} | MANAGEMENT_ROLES:
        return Response({'detail': 'Chỉ Kế toán mới duyệt HĐ.'}, status=403)

    contract = Contract.objects.filter(pk=pk, is_deleted=False).first()
    if not contract:
        return Response({'detail': 'Không tìm thấy HĐ.'}, status=404)
    if contract.approval_status != 'pending_kt':
        return Response({'detail': 'HĐ chưa được submit hoặc đã xử lý.'}, status=400)

    with transaction.atomic():
        contract.approval_status = 'approved'
        contract.approved_by = request.user
        contract.approved_at = timezone.now()
        # Nếu KH đã thanh toán khi tạo HĐ → ghi nhận
        if contract.final_amount > 0:
            contract.payment_status = 'received'
        contract.save()

        # Chuyển KH vào hàng chờ CSKH hoặc đang chăm sóc nếu đã phân CSKH
        customer = contract.customer
        customer.is_customer = True
        if customer.cskh_id:
            customer.status = 'dang_cham_soc'
        else:
            customer.status = 'cho_phan_cskh'
        customer.save(update_fields=['status', 'is_customer', 'updated_at'])

        # --- Track 2: xu ly dong SAN PHAM trong items (chi commit khi KT duyet) ---
        items = contract.items or []
        changed = False
        for item in items:
            product_id = item.get('product_id')
            if not product_id or item.get('service_id'):
                continue  # dong dich vu -> bo qua
            if item.get('delivery_status') == 'da_giao':
                continue  # da xu ly truoc do
            qty = int(item.get('qty', 1) or 1)
            is_gift = bool(item.get('is_gift'))
            delivery = item.get('delivery_status') or 'giao_ngay'
            product = Product.objects.select_for_update().filter(id=product_id).first()
            if not product:
                continue
            if is_gift:
                # hang tang: tru ton ngay, doanh thu 0, ghi chi phi khuyen mai
                product.stock = product.stock - qty
                product.save(update_fields=['stock'])
                StockMovement.objects.create(
                    product=product, change=-qty, reason='tang',
                    ref_type='contract', ref_id=contract.id,
                    cost_at_time=product.cost_price, created_by=request.user)
                item['cost_price'] = int(product.cost_price)
                item['delivery_status'] = 'da_giao'
                changed = True
            elif delivery == 'dat_truoc' or product.stock < qty:
                # hang chua ve / khong du ton -> dat truoc, cho giao, chua tru ton
                item['delivery_status'] = 'cho_giao'
                changed = True
            else:
                # co san + giao ngay -> tru ton ngay
                product.stock = product.stock - qty
                product.save(update_fields=['stock'])
                StockMovement.objects.create(
                    product=product, change=-qty, reason='ban',
                    ref_type='contract', ref_id=contract.id,
                    cost_at_time=product.cost_price, created_by=request.user)
                item['cost_price'] = int(product.cost_price)
                item['delivery_status'] = 'da_giao'
                changed = True
        if changed:
            contract.items = items
            contract.save(update_fields=['items'])

    send_notification(
        contract.created_by,
        'hd_approved',
        f'HĐ {contract.contract_no} đã được duyệt',
        body=f'KH: {contract.customer.full_name}',
        data={'contract_id': contract.id},
    )
    send_notification_to_roles(
        ['LEAD_CSKH'], 'hd_approved_cskh',
        f'HĐ {contract.contract_no} đã duyệt — KH {contract.customer.full_name} sẵn sàng chăm sóc',
        data={'contract_id': contract.id, 'customer_id': contract.customer.id},
    )
    return Response(ContractDetailSerializer(contract).data)


@api_view(['POST'])
@permission_classes([IsAuthenticated])
def deliver_contract(request, pk):
    """
    POST /api/contracts/{id}/deliver/
    Giao hang cho dong SAN PHAM dat truoc (delivery_status='cho_giao') -> tru ton, chuyen 'da_giao'.
    Body tuy chon: {"product_id": N} de giao rieng 1 san pham; khong co -> giao tat ca dong du ton.
    Quyen: Sale tao HD hoac Management.
    """
    contract = Contract.objects.filter(pk=pk, is_deleted=False).first()
    if not contract:
        return Response({'detail': 'Không tìm thấy HĐ.'}, status=404)
    if not (contract.created_by_id == request.user.id or request.user.role in MANAGEMENT_ROLES):
        return Response({'detail': 'Chỉ Sale của HĐ hoặc Quản lý mới giao hàng.'}, status=403)
    if contract.approval_status != 'approved':
        return Response({'detail': 'HĐ chưa duyệt, không thể giao hàng.'}, status=400)

    only_product = request.data.get('product_id')
    items = contract.items or []
    delivered, thieu_ton = [], []
    with transaction.atomic():
        for item in items:
            product_id = item.get('product_id')
            if not product_id or item.get('service_id'):
                continue
            if item.get('delivery_status') != 'cho_giao':
                continue
            if only_product and str(product_id) != str(only_product):
                continue
            qty = int(item.get('qty', 1) or 1)
            product = Product.objects.select_for_update().filter(id=product_id).first()
            if not product:
                continue
            if product.stock < qty:
                thieu_ton.append(item.get('name') or str(product_id))
                continue
            product.stock = product.stock - qty
            product.save(update_fields=['stock'])
            StockMovement.objects.create(
                product=product, change=-qty, reason='ban',
                ref_type='contract', ref_id=contract.id,
                cost_at_time=product.cost_price, created_by=request.user)
            item['cost_price'] = int(product.cost_price)
            item['delivery_status'] = 'da_giao'
            delivered.append(product.name)
        contract.items = items
        contract.save(update_fields=['items'])

    msg = ('Đã giao: ' + ', '.join(delivered)) if delivered else 'Không có dòng nào được giao.'
    if thieu_ton:
        msg += ' | Thiếu tồn (cần nhập kho): ' + ', '.join(thieu_ton)
    return Response({'detail': msg, 'delivered': delivered, 'thieu_ton': thieu_ton})


@api_view(['POST'])
@permission_classes([IsAuthenticated])
def reject_contract(request, pk):
    """
    POST /api/contracts/{id}/reject/
    KT từ chối HĐ. Body: {"reason": "..."}
    pending_kt → rejected
    """
    if request.user.role not in {'KE_TOAN'} | MANAGEMENT_ROLES:
        return Response({'detail': 'Chỉ Kế toán mới từ chối HĐ.'}, status=403)

    contract = Contract.objects.filter(pk=pk, is_deleted=False).first()
    if not contract:
        return Response({'detail': 'Không tìm thấy HĐ.'}, status=404)
    if contract.approval_status != 'pending_kt':
        return Response({'detail': 'HĐ chưa được submit hoặc đã xử lý.'}, status=400)

    reason = request.data.get('reason', '').strip()
    if not reason:
        return Response({'detail': 'Vui lòng nhập lý do từ chối.'}, status=400)

    contract.approval_status = 'rejected'
    contract.approved_by = request.user
    contract.approved_at = timezone.now()
    contract.notes = (contract.notes + f'\n[Từ chối: {reason}]').strip()
    contract.save()

    send_notification(
        contract.created_by,
        'hd_rejected',
        f'HĐ {contract.contract_no} bị từ chối',
        body=reason,
        data={'contract_id': contract.id},
    )
    return Response(ContractDetailSerializer(contract).data)
