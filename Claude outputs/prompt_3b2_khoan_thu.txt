===================== frontend/src/pages/sale/modals/SuaThuModal.jsx =====================
import { useState, useEffect } from 'react'
import { IconX, IconCheck } from '@tabler/icons-react'
import { editPayment } from '../../../api/payments'
import toast from 'react-hot-toast'

const overlay = { position: 'fixed', inset: 0, background: 'rgba(0,0,0,.45)', display: 'flex', alignItems: 'center', justifyContent: 'center', zIndex: 200, padding: 16 }
const fmt = n => Number(n ?? 0).toLocaleString('vi-VN')
const num = v => Number(v || 0)
const todayStr = () => { const d = new Date(); return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}` }

export default function SuaThuModal({ onClose, onDone, payment }) {
  const init = v => (Number(v || 0) > 0 ? String(Number(v)) : '')
  const [cash, setCash] = useState(init(payment?.cash_amount))
  const [transfer, setTransfer] = useState(init(payment?.transfer_amount))
  const [receivedAt, setReceivedAt] = useState(payment?.received_at || todayStr())
  const [note, setNote] = useState(payment?.note || '')
  const [saving, setSaving] = useState(false)

  useEffect(() => {
    const onKey = e => e.key === 'Escape' && onClose()
    document.addEventListener('keydown', onKey)
    return () => document.removeEventListener('keydown', onKey)
  }, [])

  const total = num(cash) + num(transfer)

  const handleSave = async () => {
    if (total <= 0) { toast.error('Nhập số tiền thu'); return }
    setSaving(true)
    try {
      await editPayment(payment.id, { cash_amount: num(cash), transfer_amount: num(transfer), received_at: receivedAt, note })
      toast.success('Đã sửa — chờ Kế toán xác nhận lại')
      onDone?.(); onClose()
    } catch (e) {
      toast.error(e.response?.data?.detail ?? 'Lỗi sửa lần thu')
    } finally { setSaving(false) }
  }

  const inputStyle = { width: '100%', border: '1px solid #dde3ef', borderRadius: 7, padding: '7px 10px', fontSize: 13, boxSizing: 'border-box', fontFamily: 'inherit' }
  const label = txt => <label style={{ fontSize: 12, fontWeight: 600, color: '#374151', display: 'block', marginBottom: 4 }}>{txt}</label>

  return (
    <div style={overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div style={{ background: '#fff', borderRadius: 12, width: '100%', maxWidth: 440, maxHeight: '92vh', overflowY: 'auto', boxShadow: '0 24px 60px rgba(0,0,0,.22)' }}>
        <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '14px 18px', borderBottom: '1px solid #f1f5f9' }}>
          <p style={{ fontWeight: 700, fontSize: 14, color: '#0f2044', margin: 0 }}>Sửa lần thu — {payment?.contract_no}</p>
          <button onClick={onClose} style={{ border: 'none', background: 'none', cursor: 'pointer', color: '#9ca3af' }}><IconX size={18} /></button>
        </div>
        <div style={{ padding: '14px 18px', display: 'flex', flexDirection: 'column', gap: 12 }}>
          {payment?.verify_note && (
            <div style={{ padding: '10px 14px', background: '#fef2f2', border: '1px solid #fecaca', borderRadius: 8 }}>
              <div style={{ fontSize: 11, color: '#991b1b', fontWeight: 700, marginBottom: 2 }}>Kế toán báo lệch</div>
              <div style={{ fontSize: 12, color: '#991b1b' }}>{payment.verify_note}</div>
            </div>
          )}
          <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 10 }}>
            <div>
              {label('Tiền mặt (đ)')}
              <input type="number" value={cash} onChange={e => setCash(e.target.value)} placeholder="0" style={inputStyle} />
            </div>
            <div>
              {label('Chuyển khoản (đ)')}
              <input type="number" value={transfer} onChange={e => setTransfer(e.target.value)} placeholder="0" style={inputStyle} />
            </div>
          </div>
          <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 10 }}>
            <div>
              {label('Ngày thu')}
              <input type="date" value={receivedAt} onChange={e => setReceivedAt(e.target.value)} style={inputStyle} />
            </div>
            <div>
              {label('Tổng thu')}
              <div style={{ padding: '7px 10px', fontSize: 16, fontWeight: 800, color: '#15803d' }}>{fmt(total)}đ</div>
            </div>
          </div>
          <div>
            {label('Ghi chú')}
            <input value={note} onChange={e => setNote(e.target.value)} placeholder="Tùy chọn..." style={inputStyle} />
          </div>
          <div style={{ display: 'flex', gap: 8, justifyContent: 'flex-end', paddingTop: 4, borderTop: '1px solid #f1f5f9' }}>
            <button onClick={onClose} style={{ padding: '7px 14px', borderRadius: 7, border: '1px solid #dde3ef', background: '#fff', fontSize: 13, cursor: 'pointer' }}>Hủy</button>
            <button onClick={handleSave} disabled={saving}
              style={{ display: 'flex', alignItems: 'center', gap: 5, padding: '7px 18px', borderRadius: 7, border: 'none', background: saving ? '#d1d5db' : '#15803d', color: '#fff', fontSize: 13, fontWeight: 700, cursor: 'pointer' }}>
              <IconCheck size={14} /> {saving ? '...' : 'Lưu & gửi lại'}
            </button>
          </div>
        </div>
      </div>
    </div>
  )
}

===================== frontend/src/pages/sale/tabs/KhoanThuTab.jsx =====================
import { useState, useEffect } from 'react'
import { IconRefresh, IconEdit } from '@tabler/icons-react'
import { getPaymentsMine } from '../../../api/payments'
import SuaThuModal from '../modals/SuaThuModal'

const fmt = n => Number(n ?? 0).toLocaleString('vi-VN')
const STATUS = {
  cho_xac_nhan: { l: 'Chờ xác nhận', c: '#854d0e', bg: '#fef9c3' },
  da_xac_nhan:  { l: 'Đã xác nhận', c: '#15803d', bg: '#dcfce7' },
  lech:         { l: 'Lệch - cần sửa', c: '#991b1b', bg: '#fee2e2' },
}

export default function KhoanThuTab() {
  const [rows, setRows] = useState([])
  const [loading, setLoading] = useState(false)
  const [filter, setFilter] = useState('all')
  const [suaTarget, setSuaTarget] = useState(null)

  const load = async () => {
    setLoading(true)
    try { const r = await getPaymentsMine(); setRows(r.data ?? []) } catch { } finally { setLoading(false) }
  }
  useEffect(() => { load() }, [])

  const lechCount = rows.filter(r => r.verify_status === 'lech').length
  const choCount = rows.filter(r => r.verify_status === 'cho_xac_nhan').length
  const filtered = filter === 'lech' ? rows.filter(r => r.verify_status === 'lech')
    : filter === 'cho' ? rows.filter(r => r.verify_status === 'cho_xac_nhan')
    : rows

  const breakdown = p => [
    Number(p.cash_amount) > 0 && `TM ${fmt(p.cash_amount)}`,
    Number(p.transfer_amount) > 0 && `CK ${fmt(p.transfer_amount)}`,
    Number(p.installment_amount) > 0 && `Trả góp ${fmt(p.installment_amount)}`,
  ].filter(Boolean).join(' · ')

  const seg = (k, l) => (
    <button key={k} onClick={() => setFilter(k)}
      style={{ padding: '4px 10px', border: `1px solid ${filter === k ? '#15803d' : '#dde3ef'}`, borderRadius: 7, background: filter === k ? '#15803d' : '#fff', color: filter === k ? '#fff' : '#64748b', fontSize: 11, fontWeight: filter === k ? 700 : 400, cursor: 'pointer', fontFamily: 'inherit', whiteSpace: 'nowrap' }}>{l}</button>
  )

  const COLS = '1.4fr 100px 1fr 90px 120px 70px'

  return (
    <div style={{ padding: '14px 16px', display: 'flex', flexDirection: 'column', gap: 14 }}>
      {lechCount > 0 && (
        <div style={{ padding: '10px 14px', background: '#fef2f2', border: '1px solid #fecaca', borderRadius: 10, fontSize: 12, color: '#991b1b', fontWeight: 700 }}>
          ⚠️ Có {lechCount} khoản bị Kế toán báo lệch — cần sửa lại.
        </div>
      )}

      <div style={{ display: 'flex', alignItems: 'center', gap: 6, flexWrap: 'wrap' }}>
        {seg('all', `Tất cả (${rows.length})`)}
        {seg('cho', `Chờ xác nhận (${choCount})`)}
        {seg('lech', `Cần sửa (${lechCount})`)}
        <button onClick={load} style={{ marginLeft: 'auto', padding: '6px 10px', border: '1px solid #dde3ef', borderRadius: 7, background: '#fff', cursor: 'pointer', color: '#64748b', display: 'flex', alignItems: 'center', gap: 4, fontSize: 12 }}>
          <IconRefresh size={14} /> Làm mới
        </button>
      </div>

      <div style={{ background: '#fff', border: '1px solid #dde3ef', borderRadius: 10, overflow: 'hidden' }}>
        <div style={{ padding: '10px 14px', borderBottom: '1px solid #f1f5f9', background: '#f8fafc', display: 'grid', gridTemplateColumns: COLS, gap: 8, fontSize: 10, fontWeight: 700, color: '#64748b' }}>
          <span>Khách / HĐ</span>
          <span style={{ textAlign: 'right' }}>Tổng thu</span>
          <span>Chi tiết</span>
          <span>Ngày</span>
          <span>Trạng thái</span>
          <span></span>
        </div>
        {loading ? (
          <div style={{ padding: 24, textAlign: 'center', color: '#9ca3af', fontSize: 12 }}>Đang tải...</div>
        ) : filtered.length === 0 ? (
          <div style={{ padding: 24, textAlign: 'center', color: '#15803d', fontSize: 12 }}>Không có khoản thu</div>
        ) : filtered.map(p => {
          const st = STATUS[p.verify_status] || { l: p.verify_status, c: '#64748b', bg: '#f1f5f9' }
          return (
            <div key={p.id} style={{ padding: '10px 14px', borderTop: '1px solid #f1f5f9', display: 'grid', gridTemplateColumns: COLS, gap: 8, alignItems: 'center' }}>
              <div>
                <div style={{ fontSize: 12, fontWeight: 700, color: '#0f2044' }}>{p.customer_name}</div>
                <div style={{ fontSize: 10, color: '#64748b' }}>{p.contract_no}</div>
              </div>
              <div style={{ fontSize: 12, textAlign: 'right', fontWeight: 700, color: '#0f2044' }}>{fmt(p.total)}đ</div>
              <div style={{ fontSize: 10, color: '#64748b' }}>{breakdown(p)}</div>
              <div style={{ fontSize: 10, color: '#64748b' }}>{(p.received_at || '').slice(0, 10)}</div>
              <div>
                <span style={{ background: st.bg, color: st.c, padding: '2px 8px', borderRadius: 10, fontSize: 10, fontWeight: 700 }}>{st.l}</span>
                {p.verify_status === 'lech' && p.verify_note && (
                  <div style={{ fontSize: 9, color: '#991b1b', marginTop: 3 }}>{p.verify_note}</div>
                )}
              </div>
              <div>
                {p.verify_status === 'lech' && (
                  <button onClick={() => setSuaTarget(p)}
                    style={{ display: 'flex', alignItems: 'center', gap: 3, justifyContent: 'center', fontSize: 11, padding: '4px 8px', borderRadius: 6, border: 'none', background: '#dc2626', color: '#fff', cursor: 'pointer', fontWeight: 600, width: '100%' }}>
                    <IconEdit size={12} /> Sửa
                  </button>
                )}
              </div>
            </div>
          )
        })}
      </div>

      {suaTarget && <SuaThuModal payment={suaTarget} onClose={() => setSuaTarget(null)} onDone={() => { setSuaTarget(null); load() }} />}
    </div>
  )
}

import { useState } from 'react'
import { useSearchParams } from 'react-router-dom'
import { IconPlus } from '@tabler/icons-react'
import useAuthStore from '../../store/authStore'
import { getUserRole } from '../../utils/rolesV2'
import AppLayout from '../../components/layout/AppLayout'
import SalePageView from './views/SalePageView'
import ChotHDModal from './modals/ChotHDModal'
import CongNoTab from './tabs/CongNoTab'

const ACCENT = '#15803d'
const ALLOWED = ['SALE', 'LEAD_SALE', 'QUAN_LY', 'CHU_DN']
const TABS = [
  { key: 'khach', label: 'Khách của tôi' },
  { key: 'congno', label: 'Công nợ & Thu nợ' },
]

export default function SalePage() {
  const user = useAuthStore(s => s.user)
  const role = getUserRole(user)
  const [showChot,  setShowChot]  = useState(false)
  const [reloadKey, setReloadKey] = useState(0)
  const [searchParams, setSearchParams] = useSearchParams()
  const VALID = TABS.map(t => t.key)
  const activeTab = VALID.includes(searchParams.get('tab')) ? searchParams.get('tab') : 'khach'
  const setActiveTab = (k) => setSearchParams({ tab: k }, { replace: true })

  if (!ALLOWED.includes(role)) {
    return (
      <AppLayout title="Sale">
        <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', height: '50vh', gap: 8 }}>
          <div style={{ fontSize: 32 }}>🔒</div>
          <div style={{ fontSize: 14, fontWeight: 700, color: '#0f2044' }}>Bạn không có quyền truy cập màn Sale</div>
          <div style={{ fontSize: 12, color: '#64748b' }}>Liên hệ quản lý để được cấp quyền</div>
        </div>
      </AppLayout>
    )
  }

  const actions = (
    <button onClick={() => setShowChot(true)}
      style={{ display: 'flex', alignItems: 'center', gap: 5, padding: '6px 14px', borderRadius: 7, border: 'none', background: ACCENT, color: '#fff', fontSize: 12, fontWeight: 700, cursor: 'pointer' }}>
      <IconPlus size={14} stroke={2.5} /> Tạo HĐ
    </button>
  )

  return (
    <AppLayout title="Sale" meta={user?.display_name ?? user?.email} actions={actions} tabs={TABS} activeTab={activeTab} onTabChange={setActiveTab}>
      {activeTab === 'khach' && <SalePageView reloadKey={reloadKey} />}
      {activeTab === 'congno' && <CongNoTab />}
      {showChot && (
        <ChotHDModal
          onClose={() => setShowChot(false)}
          onDone={() => { setReloadKey(k => k + 1); setShowChot(false) }}
        />
      )}
    </AppLayout>
  )
}

                onClick={() => setThuNoTarget({ id: c.contract_id, customer_name: c.customer_name, final_amount: c.final_amount, paid: c.paid, remaining: c.remaining })}
                style={{ fontSize: 11, padding: '4px 8px', borderRadius: 6, border: 'none', background: '#15803d', color: '#fff', cursor: 'pointer', fontWeight: 600, width: '100%' }}>
                Thu
              </button>
            </div>
          </div>
        ))}
        {rows.length > 0 && (
          <div style={{ padding: '10px 14px', borderTop: '2px solid #dde3ef', display: 'grid', gridTemplateColumns: '1fr 80px 80px 80px 70px', gap: 8 }}>
            <div style={{ fontSize: 11, fontWeight: 700, color: '#374151' }}>Tổng ({rows.length} HĐ)</div>
            <div style={{ fontSize: 11, textAlign: 'right', fontWeight: 700 }}>{fmt(rows.reduce((s, c) => s + Number(c.final_amount ?? 0), 0))}</div>
            <div style={{ fontSize: 11, textAlign: 'right', fontWeight: 700, color: '#15803d' }}>{fmt(rows.reduce((s, c) => s + Number(c.paid ?? 0), 0))}</div>
            <div style={{ fontSize: 11, textAlign: 'right', fontWeight: 800, color: '#dc2626' }}>{fmt(totalDebt)}</div>
            <div></div>
          </div>
        )}
      </div>

      {thuNoTarget && <ThuNoKTModal contract={thuNoTarget} onClose={() => setThuNoTarget(null)} onDone={() => { setThuNoTarget(null); load() }} />}
    </div>
  )
}
